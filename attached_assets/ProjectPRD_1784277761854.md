from fastapi import APIRouter, HTTPException, Depends, status
from fastapi.security import OAuth2PasswordBearer, OAuth2PasswordRequestForm
from datetime import datetime, timedelta
from jose import JWTError, jwt
from passlib.context import CryptContext
from app.models import User, UserCreate, UserLogin, Token, TokenData
from app.database import get_database
from app.config import settings
from bson import ObjectId

router = APIRouter()
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")
oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/api/auth/login")

def verify_password(plain_password, hashed_password):
    return pwd_context.verify(plain_password, hashed_password)

def get_password_hash(password):
    return pwd_context.hash(password)

def create_access_token(data: dict, expires_delta: timedelta = None):
    to_encode = data.copy()
    if expires_delta:
        expire = datetime.utcnow() + expires_delta
    else:
        expire = datetime.utcnow() + timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)
    to_encode.update({"exp": expire})
    encoded_jwt = jwt.encode(to_encode, settings.SECRET_KEY, algorithm=settings.ALGORITHM)
    return encoded_jwt

async def get_current_user(token: str = Depends(oauth2_scheme)):
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        payload = jwt.decode(token, settings.SECRET_KEY, algorithms=[settings.ALGORITHM])
        email: str = payload.get("sub")
        if email is None:
            raise credentials_exception
        token_data = TokenData(email=email)
    except JWTError:
        raise credentials_exception
    
    db = get_database()
    user = await db.users.find_one({"email": token_data.email})
    if user is None:
        raise credentials_exception
    return user

@router.post("/register", response_model=dict)
async def register(user: UserCreate):
    db = get_database()
    
    # Check if user exists
    existing_user = await db.users.find_one({"email": user.email})
    if existing_user:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Email already registered"
        )
    
    # Create new user
    hashed_password = get_password_hash(user.password)
    new_user = {
        "email": user.email,
        "username": user.username,
        "hashed_password": hashed_password,
        "created_at": datetime.utcnow(),
        "updated_at": datetime.utcnow()
    }
    
    result = await db.users.insert_one(new_user)
    return {"message": "User created successfully", "user_id": str(result.inserted_id)}

@router.post("/login", response_model=Token)
async def login(form_data: OAuth2PasswordRequestForm = Depends()):
    db = get_database()
    user = await db.users.find_one({"email": form_data.username})
    
    if not user or not verify_password(form_data.password, user["hashed_password"]):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Incorrect email or password",
            headers={"WWW-Authenticate": "Bearer"},
        )
    
    access_token = create_access_token(data={"sub": user["email"]})
    return {"access_token": access_token, "token_type": "bearer"}

from app.config import settings
from openai import OpenAI
import pandas as pd
import numpy as np

from typing import List, Dict

router = APIRouter()
client = OpenAI(api_key=settings.OPENAI_API_KEY)

async def analyze_spending_patterns(user_id: str) -> Dict:
    db = get_database()
    
    # Get last 30 days of expenses
    thirty_days_ago = datetime.utcnow() - timedelta(days=30)
    cursor = db.expenses.find({
        "user_id": user_id,
        "date": {"$gte": thirty_days_ago}
    })
    
    expenses = []
    async for expense in cursor:
        expenses.append(expense)
    
    if not expenses:
        return {"total_spent": 0, "categories": {}, "avg_daily": 0}
    
    # Create DataFrame for analysis
    df = pd.DataFrame(expenses)
    total_spent = df["amount"].sum()
    
    # Category breakdown
    category_spending = df.groupby("category")["amount"].sum().to_dict()
    avg_daily = total_spent / 30
    
    # Detect anomalies (expenses > 2 standard deviations from mean)
    mean_spending = df["amount"].mean()
    std_spending = df["amount"].std()
    anomalies = df[df["amount"] > (mean_spending + 2 * std_spending)]
    
    return {
        "total_spent": total_spent,
        "categories": category_spending,
        "avg_daily": avg_daily,
        "top_category": max(category_spending, key=category_spending.get) if category_spending else None,
        "anomalies": anomalies.to_dict("records") if not anomalies.empty else []
    }

async def get_budget_comparison(user_id: str) -> Dict:
    db = get_database()
    current_month = datetime.utcnow().month
    current_year = datetime.utcnow().year
    
    # Get budgets
    budgets = await db.budgets.find({
        "user_id": user_id,
        "month": current_month,
        "year": current_year
    }).to_list(length=None)
    
    # Get actual spending this month
    month_start = datetime(current_year, current_month, 1)
    cursor = db.expenses.find({
        "user_id": user_id,
        "date": {"$gte": month_start}
    })
    
    actual_spending = {}
    async for expense in cursor:
        category = expense["category"]
        actual_spending[category] = actual_spending.get(category, 0) + expense["amount"]
    
    return {
        "budgets": budgets,
        "actual": actual_spending
    }

@router.post("/advice", response_model=AIAdviceResponse)
async def get_ai_advice(request: AIAdviceRequest, current_user: dict = Depends(get_current_user)):
    try:
        # Get user's financial data
        spending_patterns = await analyze_spending_patterns(str(current_user["_id"]))
        budget_data = await get_budget_comparison(str(current_user["_id"]))
        
        # Create context for AI
        context = f"""
        User's financial summary for the last 30 days:
        - Total spent: ${spending_patterns['total_spent']:.2f}
        - Average daily spending: ${spending_patterns['avg_daily']:.2f}
        - Top spending category: {spending_patterns.get('top_category', 'None')}
        - Category breakdown: {spending_patterns.get('categories', {})}
        
        Budget status:
        - Budgets set: {budget_data.get('budgets', [])}
        - Actual spending: {budget_data.get('actual', {})}
        
        User question: {request.user_query}
        """
        
        # Call OpenAI API
        response = client.chat.completions.create(
            model="gpt-4",
            messages=[
                {"role": "system", "content": "You are an expert financial advisor helping users manage their personal finances. Provide actionable, personalized advice based on their spending patterns."},
                {"role": "user", "content": context}
            ],
            temperature=0.7,
            max_tokens=500
        )
        
        advice = response.choices[0].message.content
        
        # Generate suggestions
        suggestions = []
        if spending_patterns['total_spent'] > 0:
            top_category = spending_patterns.get('top_category')
            if top_category and top_category in spending_patterns['categories']:
                category_spend = spending_patterns['categories'][top_category]
                suggestions.append(f"Consider reducing your {top_category} spending (${category_spend:.2f}) by 10% to save ${category_spend * 0.1:.2f} per month")
            
            if spending_patterns['avg_daily'] > 100:
                suggestions.append("Try to reduce your daily average spending below $100 to improve your savings")
        
        return AIAdviceResponse(
            response=advice,
            suggestions=suggestions if suggestions else None
        )
        
    except Exception as e:
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail=f"AI service error: {str(e)}"
        )

    try:
        db = get_database()
        
        # Get last 90 days of expenses
        ninety_days_ago = datetime.utcnow() - timedelta(days=90)
        cursor = db.expenses.find({
            "user_id": str(current_user["_id"]),
            "date": {"$gte": ninety_days_ago}
        })
        
        expenses = []
        async for expense in cursor:
            expenses.append(expense)
        
        if not expenses:
            return {"prediction": "Not enough data for predictions", "next_month_estimate": 0}
        
        # Simple linear regression for prediction
        df = pd.DataFrame(expenses)
        df["day"] = (df["date"] - df["date"].min()).dt.days
        
        # Group by day
        daily_spending = df.groupby("day")["amount"].sum().reset_index()
        
        if len(daily_spending) < 2:
            return {"prediction": "Not enough data points", "next_month_estimate": 0}
        
        # Simple linear model
        X = daily_spending["day"].values.reshape(-1, 1)
        y = daily_spending["amount"].values
        
        from sklearn.linear_model import LinearRegression
        model = LinearRegression()
        model.fit(X, y)
        
        # Predict next 30 days
        next_30_days = np.arange(daily_spending["day"].max() + 1, daily_spending["day"].max() + 31).reshape(-1, 1)
        predictions = model.predict(next_30_days)
        
        next_month_estimate = max(0, predictions.sum())
        
        return {
            "prediction": f"Based on your spending patterns, you'll likely spend approximately ${next_month_estimate:.2f} next month",
            "next_month_estimate": round(next_month_estimate, 2),
            "confidence": "Medium" if len(daily_spending) > 15 else "Low"
        }
        
    except Exception as e:
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail=f"Prediction error: {str(e)}"
        )

    period: str = "monthly",
    current_user: dict = Depends(get_current_user)
):
    db = get_database()
    
    # Get expenses based on period
    if period == "weekly":
        start_date = datetime.utcnow() - timedelta(days=7)
    elif period == "monthly":
        start_date = datetime.utcnow() - timedelta(days=30)
    elif period == "yearly":
        start_date = datetime.utcnow() - timedelta(days=365)
    else:
        start_date = datetime.utcnow() - timedelta(days=30)
    
    cursor = db.expenses.find({
        "user_id": str(current_user["_id"]),
        "date": {"$gte": start_date}
    }).sort("date", 1)
    
    expenses = []
    async for expense in cursor:
        expenses.append(expense)
    
    if not expenses:
        return {"trends": [], "summary": {}}
    
    # Create DataFrame
    df = pd.DataFrame(expenses)
    df["date"] = pd.to_datetime(df["date"])
    
    # Group by date
    if period == "weekly":
        df["period"] = df["date"].dt.strftime("%Y-W%W")
    elif period == "monthly":
        df["period"] = df["date"].dt.strftime("%Y-%m")
    else:  # yearly
        df["period"] = df["date"].dt.strftime("%Y")
    
    grouped = df.groupby("period")["amount"].sum().reset_index()
    
    trends = grouped.to_dict("records")
    
    # Calculate statistics
    mean_spending = df["amount"].mean()
    std_spending = df["amount"].std()
    
    return {
        "trends": trends,
        "summary": {
            "total": round(df["amount"].sum(), 2),
            "average": round(mean_spending, 2),
            "std_dev": round(std_spending, 2),
            "min": round(df["amount"].min(), 2),
            "max": round(df["amount"].max(), 2),
            "count": len(df)
        }
    }
